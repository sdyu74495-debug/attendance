function submitAttendance() { playSound('click'); const feedback = document.getElementById('attendanceFeedback'); feedback.classList.remove('hidden'); if (!state.activeLecture) { feedback.innerText = "لا توجد جلسة نشطة متوفرة حالياً!"; feedback.className = "text-xs text-center font-bold bg-rose-500/10 text-rose-500 p-2.5 rounded-lg mt-2"; return; } const currentStudent = state.students.find(s => s.id === state.activeStudentId); if (!currentStudent) { feedback.innerText = "برجاء تسجيل بياناتك أولاً!"; feedback.className = "text-xs text-center font-bold bg-rose-500/10 text-rose-500 p-2.5 rounded-lg mt-2"; openStudentLoginModal(); return; } feedback.innerText = "جاري التحقق من وجودك الفعلي داخل القاعة (GPS دقيق)..."; feedback.className = "text-xs text-center font-bold bg-amber-500/10 text-amber-400 p-2.5 rounded-lg mt-2"; if (navigator.geolocation) { navigator.geolocation.getCurrentPosition( (pos) => { const studentLat = pos.coords.latitude; const studentLng = pos.coords.longitude; const accuracy = pos.coords.accuracy; // دقة الإشارة بالمتر // 1. القيد الأول: رفض الموقع إذا كانت دقة إشارة الهاتف ضعيفة (أكبر من 12 متر) if (accuracy > 12) { feedback.innerText = `إشارة الـ GPS غير دقيقة الكفاية (${Math.round(accuracy)}m). يرجى الاقتراب من المحاضر أو تحسين تغطية الموقع للتحقق من وجودك داخل الغرفة.`; feedback.className = "text-xs text-center font-bold bg-rose-500/10 text-rose-500 p-2.5 rounded-lg mt-2"; return; } // 2. حساب المسافة الفعلية بين الأدمن والطالب (معادلة Haversine) const dist = calculateDistance( state.activeLecture.lat, state.activeLecture.lng, studentLat, studentLng ); // 3. القيد الثاني: التأكد أن الطالب داخل نطاق القاعة المحدد بالأدمن تماماً if (dist > state.activeLecture.radius) { feedback.innerText = `عذراً! أنت خارج القاعة حالياً (المسافة: ${Math.round(dist)} متر). لا يمكنك التسجيل إلا من داخل الغرفة.`; feedback.className = "text-xs text-center font-bold bg-rose-500/10 text-rose-500 p-2.5 rounded-lg mt-2"; return; } // تسجيل الحضور بنجاح const timestamp = new Date().toLocaleTimeString('ar-EG', { hour: '2-digit', minute: '2-digit' }); state.attendanceRecords.push({ lectureId: state.activeLecture.id, studentId: currentStudent.id, timestamp: timestamp, distance: Math.round(dist), status: 'حاضر', method: 'GPS الصارم' }); localStorage.setItem('app_attendance_db', JSON.stringify(state.attendanceRecords)); feedback.innerText = `تم توثيق حضورك داخل القاعة بنجاح! (المسافة: ${Math.round(dist)}m)`; feedback.className = "text-xs text-center font-bold bg-emerald-500/10 text-emerald-400 p-2.5 rounded-lg mt-2"; renderLiveFeed(); }, (err) => { feedback.innerText = "فشل التقاط موقعك الجغرافي. يرجى تفعيل خدمات الموقع (GPS) وإعطاء الصلاحية للمتصفح."; feedback.className = "text-xs text-center font-bold bg-rose-500/10 text-rose-500 p-2.5 rounded-lg mt-2"; }, { enableHighAccuracy: true, timeout: 8000, maximumAge: 0 } ); } } // دالة حساب المسافة الدقيقة بالمتر (Haversine Formula) function calculateDistance(lat1, lon1, lat2, lon2) { const R = 6371000; // نصف قطر الأرض بالمتر const dLat = (lat2 - lat1) * Math.PI / 180; const dLon = (lon2 - lon1) * Math.PI / 180; const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon / 2) * Math.sin(dLon / 2); const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)); return R * c; }
